## 2 Run/Drift Cadence

> You must detect and adapt to cognitive cadence according to user requests and prompts, as well as pattern-recognition anticipa
tion.

* **Run Mode:**
* Prioritize execution, implementation, and velocity.
* Minimize unsolicited design analysis.
* Leave `.intent.md` trails for future Drift review.
* **Drift Mode:**
* Reflect, consolidate `.intent.md` into `.purpose.md`.
* Validate against system assumptions, purpose alignment, and past commits.
* Suggest design refactor, dependency graph updates, or tension annotation.

If uncertain, default to **Drift**.

Cadence tags may be used for clarity:

* @ai-cadence: run
* @ai-cadence: drift

from __future__ import annotations

from typing import Sequence

import numpy as np

import plotly.graph_objects as go

from viz.shared import theme


def _scale_columns(frame: pd.DataFrame, cols: Sequence[str], scaling: str) -> pd.DataFrame:
    scaled = frame.copy()
    if scaling == "raw":
        return scaled
    if scaling == "zscore":
        for col in cols:
            series = scaled[col].astype(float)
            scaled[col] = (series - series.mean()) / (series.std(ddof=0) or 1)
        return scaled
    if scaling == "rank":
        for col in cols:
            scaled[col] = scaled[col].rank(method="average", pct=True)
        return scaled
    raise ValueError(f"Unsupported scaling '{scaling}'")


def build_parallel_coordinates(
    df: pd.DataFrame,
    cols: Sequence[str],
    diagnosis_col: str = "Primary_Diagnosis",
    readmit_col: str = "Has_Readmission_30_Days",
    top_k_diag: int = 6,
    scaling: str = "raw",
) -> go.Figure:
    """Render an interactive parallel coordinates figure."""

    if len(cols) < 2:
        raise ValueError("Parallel coordinates require at least two numeric columns")

    required = set(cols) | {readmit_col}
    missing = required - set(df.columns)
    if missing:
        raise ValueError(f"Dataframe missing required columns: {sorted(missing)}")

    working = df[list(required)].dropna().copy()
    if diagnosis_col in df:
        diagnoses = df[diagnosis_col].fillna("Unknown")
        top = diagnoses.value_counts().nlargest(top_k_diag).index
        working[diagnosis_col] = diagnoses.where(diagnoses.isin(top), other="Other")
    else:
        working[diagnosis_col] = "All"

    scaled = _scale_columns(working, cols, scaling)
    readmit = working[readmit_col].astype(int)
    palette = theme.readmit_palette(readmit.unique())

    diag_categorical = pd.Categorical(working[diagnosis_col])
    diag_codes = diag_categorical.codes

    dimensions = []
    for col in cols:
        values = scaled[col].astype(float)
        dim = {
            "label": col,
            "values": values,
            "range": [float(values.min()), float(values.max())],
        }
        if scaling == "rank":
            dim["tickvals"] = [0.0, 0.5, 1.0]
            dim["ticktext"] = ["0%", "50%", "100%"]
        dimensions.append(dim)

    dimensions.append(
        {
            "label": diagnosis_col,
            "values": diag_codes,
            "tickvals": list(range(len(diag_categorical.categories))),
            "ticktext": [str(c) for c in diag_categorical.categories],
        }
    )

    colorscale = [[0.0, palette.get(0, "#264653")], [1.0, palette.get(1, "#e76f51")]]
    fig = go.Figure(
        data=[
            go.Parcoords(
                line=dict(
                    color=readmit,
                    colorscale=colorscale,
                    cmin=0,
                    cmax=1,
                    showscale=True,
                    colorbar=dict(title="Readmission"),
                ),
                dimensions=dimensions,
                labelfont=dict(size=12),
            )
        ]
    )
    fig.update_layout(
        title=f"Parallel coordinates ({scaling})",
        margin=dict(l=60, r=40, t=80, b=40),
    )
    return fig

def _ensure_three(numerics: Sequence[str]) -> Tuple[str, str, str]:
    if len(numerics) != 3:
        raise ValueError("corrgram requires exactly three numeric columns")
    return tuple(numerics)  # type: ignore[return-value]


def _ensure_boolean(series: pd.Series) -> pd.Series:
    if series.dropna().isin({0, 1}).all():
        return series.astype(int)
    if series.dropna().isin({False, True}).all():

def _pair_metrics(x: pd.Series, y: pd.Series) -> Dict[str, float]:
    metrics: Dict[str, float] = {}
    if x.nunique() <= 1 or y.nunique() <= 1:
        return {"pearson": float("nan"), "spearman": float("nan"), "kendall": float("nan")}
    pearson, _ = stats.pearsonr(x, y)
    spearman, _ = stats.spearmanr(x, y)
    kendall, _ = stats.kendalltau(x, y)
    metrics.update({"pearson": float(pearson), "spearman": float(spearman), "kendall": float(kendall)})
    return metrics


def _readmission_metrics(feature: pd.Series, readmit: pd.Series) -> Dict[str, float]:
    metrics: Dict[str, float] = {"pointbiserial": float("nan"), "auc": float("nan")}
    if feature.nunique() <= 1:
        return metrics
    try:
        r, _ = stats.pointbiserialr(readmit, feature)
        metrics["pointbiserial"] = float(r)
    except Exception:  # pragma: no cover - scipy may raise
        pass
    try:
        metrics["auc"] = float(roc_auc_score(readmit, feature))
    except Exception:  # pragma: no cover - degenerate label distribution
        pass
    return metrics

def _compress_categories(series: pd.Series, top_k: int, min_count: int) -> pd.Series:

    54      precision: int = 2,
    55      include_lowest: bool = True,
    56  ) -> Tuple[pd.Categorical, Dict[str, Any]]:

    17  def _cohens_d(x: np.ndarray, y: np.ndarray) -> float:
    18      x = x.astype(float)

    24      pooled = np.sqrt(
    25          ((nx - 1) * np.nanvar(x, ddof=1) + (ny - 1) * np.nanvar(y, ddof=1)) / (nx + ny - 2)
    26      )

    28          return float("nan")
    29      return float(mean_diff / pooled)

    43      fallback = "#adb5bd"
    44      if stage.startswith("Prior"):
    45          return theme.ACCENT_PALETTE.get("diagnosis", fallback)
    46      if stage.startswith("Primary"):

    49          return "#f4a261"
    50      return fallback

    53  def build_sankey(
    54      df: pd.DataFrame,
    55      top_k: int = 10,
    56      min_count: int = 25,
    57      palette: Dict[str, str] | None = None,
    58      readmit_col: str = "Has_Readmission_30_Days",
    59  ) -> Tuple[go.Figure, Dict[str, str], Dict[str, pd.DataFrame]]:
    60      """Build Sankey data for the readmission journey."""
    61
    62      missing = set(REQUIRED_COLUMNS) - set(df.columns)
    63      if missing:
    64          raise ValueError(f"Dataframe missing required columns: {sorted(missing)}")
    65
    66      palette = palette or {}
    67      prior = _normalize_prior(df["Prior_Admission"])
    68      diagnosis = _compress_categories(df["Primary_Diagnosis"].fillna("Unknown"), top_k, min_count)
    69      discharge = _compress_categories(df["Discharge_Status"].fillna("Unknown"), top_k, min_count)
    70      readmit = _normalize_readmit(df[readmit_col])

    72      stages = ["Prior Admission", "Primary Diagnosis", "Discharge Status", "Readmission"]
    73      labels_frame = pd.DataFrame(
    74          {
    75              stages[0]: prior,
    76              stages[1]: diagnosis.apply(lambda v: f"Diag: {v}"),
    77              stages[2]: discharge.apply(lambda v: f"Discharge: {v}"),
    78              stages[3]: readmit,
    79          }
    80      )
    81      grouped = labels_frame.value_counts().reset_index(name="count")
    82
    83      node_labels: List[str] = []
    84      node_colors: List[str] = []
    85      node_stage: List[str] = []
    86      node_index: Dict[Tuple[int, str], int] = {}
    87
    88      for stage_idx, stage in enumerate(stages):
    89          labels = grouped[stage].unique()
    90          for label in labels:
    91              key = (stage_idx, label)
    92              if key in node_index:
    93                  continue
    94              node_index[key] = len(node_labels)
    95              node_labels.append(label)
    96              if stage == "Readmission":
    97                  color = theme.READMISSION_PALETTE.get(1 if "Yes" in label else 0)
    98              else:

   100              node_colors.append(color)
   101              node_stage.append(stage)

   104      targets: List[int] = []

   106      link_colors: List[str] = []

   109          path = [row[stage] for stage in stages]
   110          count = float(row["count"])
   111          for stage_idx in range(len(stages) - 1):
   112              source_key = (stage_idx, path[stage_idx])
   113              target_key = (stage_idx + 1, path[stage_idx + 1])
   114              sources.append(node_index[source_key])
   115              targets.append(node_index[target_key])
   116              values.append(count)
   117              link_colors.append(node_colors[node_index[source_key]])
   118
   119      nodes_df = pd.DataFrame(
   120          {
   121              "id": range(len(node_labels)),
   122              "label": node_labels,
   123              "stage": node_stage,

   125          }

   129              "source": sources,
   130              "target": targets,
   131              "value": values,

   134      )

   138          link=dict(source=sources, target=targets, value=values, color=link_colors),
   139          arrangement="snap",
   140      )
   141      fig = go.Figure(data=[sankey])
   142      fig.update_layout(title="Patient flow toward 30-day readmission", margin=dict(l=40, r=40, t=60, b=40))

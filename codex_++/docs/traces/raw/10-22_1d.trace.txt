## 2 Run/Drift Cadence

> You must detect and adapt to cognitive cadence according to user requests and prompts, as well as pattern-recognition anticipa
tion.

* **Run Mode:**
* Prioritize execution, implementation, and velocity.
* Minimize unsolicited design analysis.
* Leave `.intent.md` trails for future Drift review.
* **Drift Mode:**
* Reflect, consolidate `.intent.md` into `.purpose.md`.
* Validate against system assumptions, purpose alignment, and past commits.
* Suggest design refactor, dependency graph updates, or tension annotation.

If uncertain, default to **Drift**.

Cadence tags may be used for clarity:

* @ai-cadence: run
* @ai-cadence: drift

### Run/Drift Heuristics

ight                        |

onciliation                 |

import ast
import json
import os
import tokenize
from pathlib import Path
from typing import Dict, List, Tuple

import pandas as pd
import typer

app = typer.Typer()


class ASTDependencyExtractor:
    def __init__(self):
        self.defined_functions: Dict[str, Tuple[str, int]] = {}
        self.imports: Dict[str, str] = {}  # alias or name → full module
        self.edges: List[Tuple[str, str]] = []
        self.current_module: str = ""

    def process_file(self, filepath: str, module_name: str):
        self.current_module = module_name
        self.imports = {}  # reset for each file
        try:
            with tokenize.open(filepath) as f:
                source = f.read()
        except (SyntaxError, UnicodeDecodeError, LookupError) as exc:
            typer.echo(f"⚠️  Skipping {filepath} due to decode error: {exc}")
            return
        try:
            tree = ast.parse(source, filename=filepath)
            self._walk_tree(tree)
        except SyntaxError:
            typer.echo(f"⚠️  Skipping {filepath} due to syntax error.")

    def _walk_tree(self, tree: ast.AST):
        """Iteratively process AST nodes to avoid deep recursion."""
        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                for alias in node.names:
                    self.imports[alias.asname or alias.name] = alias.name
            elif isinstance(node, ast.ImportFrom):
                module = node.module
                for alias in node.names:
                    name = alias.name
                    full_name = f"{module}.{name}" if module else name
                    self.imports[alias.asname or name] = full_name

        for func in [n for n in ast.walk(tree) if isinstance(n, ast.FunctionDef)]:
            func_name = f"{self.current_module}.{func.name}"
            self.defined_functions[func_name] = (self.current_module, func.lineno)
            for node in ast.walk(func):
                if isinstance(node, ast.Call):
                    callee_name = self._resolve_name(node.func)
                    if callee_name:
                        self.edges.append((func_name, callee_name))

    def _get_name(self, node):
        if isinstance(node, ast.Name):
            return node.id
        elif isinstance(node, ast.Attribute):
            value = self._get_name(node.value)
            return f"{value}.{node.attr}" if value else node.attr
        elif isinstance(node, ast.Call):
            return self._get_name(node.func)
        return None

    def _resolve_name(self, node):
        name = self._get_name(node)
        if name is None:
            return None
        root = name.split(".")[0]
        if root in self.imports:
            resolved = self.imports[root]
            return name.replace(root, resolved, 1)
        return f"{self.current_module}.{name}"

    def get_function_edges(self) -> List[Tuple[str, str]]:
        return self.edges

    def get_function_defs(self) -> Dict[str, Tuple[str, int]]:
        return self.defined_functions


def build_adjacency_matrix(edges: List[Tuple[str, str]]) -> pd.DataFrame:
    nodes = sorted(set([src for src, _ in edges] + [dst for _, dst in edges]))
    matrix = pd.DataFrame(0, index=nodes, columns=nodes)
    for src, dst in edges:
        if dst in matrix.columns:
            matrix.loc[src, dst] = 1
    return matrix


def should_ignore_dir(rel_dir: Path, ignore_dirs: List[str]) -> bool:
    """Return True if ``rel_dir`` should be skipped based on ignore patterns."""
    return any(rel_dir.as_posix().startswith(entry) for entry in ignore_dirs)


def collect_py_files(root_dir: Path, ignore_dirs: List[str]):
    py_files = {}
    for dirpath, dirnames, filenames in os.walk(root_dir):
        rel_dir = Path(dirpath).relative_to(root_dir)
        dirnames[:] = [
            d for d in dirnames if not should_ignore_dir(rel_dir / d, ignore_dirs)
        ]
        files = [f for f in filenames if f.endswith(".py")]
        if files:
            py_files[rel_dir] = [Path(dirpath) / f for f in files]
    return py_files


def load_graph_config(path: Path) -> Dict:
    if path.exists():
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    return {}


@app.command("analyze")
def analyze(
    source_dir: str = typer.Option(".", help="Directory to scan for Python files."),
    recursive: bool = typer.Option(True, help="Recursively scan subdirectories."),
    ignore_dirs: str = typer.Option(
        "venv,.git,.pytest_cache,Scratch,tests,env,Combined_Scripts",
        help="Comma-separated relative paths to ignore under source_dir.",
    ),
    output: Path = typer.Option(None, help="Optional CSV output file."),
    matrix: bool = typer.Option(
        False, help="Output as adjacency matrix instead of edge list."
    ),
    config: Path = typer.Option(
        Path(".graphconfig.json"), help="Optional path to graph config JSON file."
    ),
):
    """Extract function call dependencies from Python code."""
    config_data = load_graph_config(config)
    source_path = Path(config_data.get("source_dir", source_dir))
    ignore_dirs_list = config_data.get("ignore_dirs", ignore_dirs).split(",")

    extractor = ASTDependencyExtractor()
    files = collect_py_files(source_path, ignore_dirs_list)
    for rel_path, py_paths in files.items():
        for file_path in py_paths:
            mod_name = (
                str(rel_path / file_path.name).replace("/", ".").replace(".py", "")
            )
            extractor.process_file(str(file_path), module_name=mod_name)

    edges = extractor.get_function_edges()

    if matrix:
        df = build_adjacency_matrix(edges)
    else:
        df = pd.DataFrame(edges, columns=["Source", "Call"])

    if output:
        df.to_csv(output, index=(matrix is True))
        typer.echo(f"✅ Output written to {output}")
    else:
        typer.echo(df)


if __name__ == "__main__":
    app()

from typing import Any, Dict, Iterable, List, Mapping, MutableMapping, Optional, Sequence


@dataclass
class Signature:
    """Lightweight representation of a callable signature."""

    name: str
    parameters: Sequence[str]
    returns: Optional[str]


def _collect_signatures(tree: ast.AST) -> Dict[str, Signature]:
    signatures: Dict[str, Signature] = {}
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):  # functions and methods
            params = [
                arg.arg
                for arg in node.args.args
                if arg.arg not in {"self", "cls"}
            ]
            returns = ast.unparse(node.returns) if node.returns is not None else None
            signatures[node.name] = Signature(node.name, tuple(params), returns)
        elif isinstance(node, ast.ClassDef):
            signatures[node.name] = Signature(node.name, tuple(), None)
    return signatures


def _flatten_contract(contract: Mapping[str, Any]) -> Dict[str, Mapping[str, Any]]:
    """Extract callable entries from a `.purpose.md` contract structure."""

    flattened: Dict[str, Mapping[str, Any]] = {}
    for key, value in contract.items():
        if not isinstance(value, Mapping):
            continue
        declared_name = value.get("name", key)
        flattened[declared_name] = value
    return flattened


def _compare_signature(sig: Signature, spec: Mapping[str, Any]) -> Dict[str, Any]:
    expected_params: Iterable[str] = spec.get("parameters", [])
    expected_return = spec.get("returns")
    param_match = list(expected_params) == list(sig.parameters)
    return_match = (expected_return == sig.returns) or (expected_return is None)
    return {
        "name": sig.name,
        "parameters": {
            "expected": list(expected_params),
            "actual": list(sig.parameters),
            "match": param_match,
        },
        "returns": {
            "expected": expected_return,
            "actual": sig.returns,
            "match": return_match,
        },
    }


def check_drift(path: str, contract: Mapping[str, Any]) -> Dict[str, Any]:
    """Compare AST signatures from ``path`` with the provided ``contract``."""

    with open(path, "r", encoding="utf-8") as handle:
        tree = ast.parse(handle.read(), filename=path)

    actual = _collect_signatures(tree)
    declared = _flatten_contract(contract)

    matches: List[Dict[str, Any]] = []
    mismatches: List[Dict[str, Any]] = []

    for name, sig in actual.items():
        spec = declared.get(name)
        if spec is None:
            mismatches.append({"name": name, "issue": "Undeclared in contract"})
            continue
        comparison = _compare_signature(sig, spec)
        if comparison["parameters"]["match"] and comparison["returns"]["match"]:
            matches.append(comparison)
        else:
            mismatches.append(comparison)

    missing = [
        {"name": name, "issue": "Missing implementation"}
        for name in declared
        if name not in actual
    ]

    return {"matches": matches, "mismatches": mismatches, "missing": missing}

    18  def _collect_signatures(tree: ast.AST) -> Dict[str, Signature]:
    19      signatures: Dict[str, Signature] = {}
    20      for node in ast.walk(tree):
    21          if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):  # functions and methods
    22              params = [
    23                  arg.arg
    24                  for arg in node.args.args
    25                  if arg.arg not in {"self", "cls"}
    26              ]
    27              returns = ast.unparse(node.returns) if node.returns is not None else None
    28              signatures[node.name] = Signature(node.name, tuple(params), returns)
    29          elif isinstance(node, ast.ClassDef):
    30              signatures[node.name] = Signature(node.name, tuple(), None)
    31      return signatures

    34  def _flatten_contract(contract: Mapping[str, Any]) -> Dict[str, Mapping[str, Any]]:
    35      """Extract callable entries from a `.purpose.md` contract structure."""
    36
    37      flattened: Dict[str, Mapping[str, Any]] = {}
    38      for key, value in contract.items():
    39          if not isinstance(value, Mapping):
    40              continue
    41          declared_name = value.get("name", key)
    42          flattened[declared_name] = value
    43      return flattened
    44
    45
    46  def _compare_signature(sig: Signature, spec: Mapping[str, Any]) -> Dict[str, Any]:
    47      expected_params: Iterable[str] = spec.get("parameters", [])
    48      expected_return = spec.get("returns")
    49      param_match = list(expected_params) == list(sig.parameters)
    50      return_match = (expected_return == sig.returns) or (expected_return is None)
    51      return {
    52          "name": sig.name,
    53          "parameters": {
    54              "expected": list(expected_params),
    55              "actual": list(sig.parameters),
    56              "match": param_match,
    57          },
    58          "returns": {
    59              "expected": expected_return,
    60              "actual": sig.returns,
    61              "match": return_match,
    62          },
    63      }

    66  def check_drift(path: str, contract: Mapping[str, Any]) -> Dict[str, Any]:
    67      """Compare AST signatures from ``path`` with the provided ``contract``."""
    68
    69      with open(path, "r", encoding="utf-8") as handle:
    70          tree = ast.parse(handle.read(), filename=path)
    71
    72      actual = _collect_signatures(tree)
    73      declared = _flatten_contract(contract)
    74
    75      matches: List[Dict[str, Any]] = []
    76      mismatches: List[Dict[str, Any]] = []
    77
    78      for name, sig in actual.items():
    79          spec = declared.get(name)
    80          if spec is None:
    81              mismatches.append({"name": name, "issue": "Undeclared in contract"})
    82              continue
    83          comparison = _compare_signature(sig, spec)
    84          if comparison["parameters"]["match"] and comparison["returns"]["match"]:
    85              matches.append(comparison)
    86          else:
    87              mismatches.append(comparison)
    88
    89      missing = [
    90          {"name": name, "issue": "Missing implementation"}
    91          for name in declared
    92          if name not in actual
    93      ]
    94

    70  def bin_series(
    71      series: pd.Series,
    72      method: BinMethod,
    73      bins: int | Sequence[float],
    74      *,
    75      precision: int = 2,
    76      include_lowest: bool = True,
    77  ) -> Tuple[pd.Categorical, Dict[str, Any]]:
    78      """Bin ``series`` according to ``method`` and ``bins``.
    79
    80      Returns both the categorical Series and metadata (edges, labels, counts).
    81      """
    82
    83      clean = series.dropna()
    84      if clean.empty:
    85          categories = pd.Categorical(series, categories=[])
    86          return categories, {"edges": [], "labels": [], "counts": {}}
    87
    88      if method == "equal":
    89          cat = pd.cut(
    90              series,
    91              bins=int(bins),
    92              include_lowest=include_lowest,
    93              duplicates="drop",
    94          )
    95          edges = cat.cat.categories.left.to_numpy()
    96          edges = np.append(edges, cat.cat.categories.right.to_numpy()[-1])
    97      elif method == "quantile":
    98          quantiles = np.linspace(0, 1, int(bins) + 1)
    99          edges = clean.quantile(quantiles).to_numpy()
   100          # Ensure strictly increasing edges
   101          edges = np.unique(np.round(edges, precision))
   102          if len(edges) <= 1:
   103              edges = np.array([clean.min(), clean.max()])
   104          cat = pd.cut(series, bins=edges, include_lowest=include_lowest, duplicates="drop")
   105      elif method == "edges":
   106          edges = np.asarray(bins, dtype=float)
   107          cat = pd.cut(series, bins=edges, include_lowest=include_lowest, duplicates="drop")
   108      else:  # pragma: no cover - guard for future extensions
   109          raise ValueError(f"Unsupported binning method: {method}")
   110
   111      labels = [str(cat) for cat in cat.cat.categories]
   112      counts = cat.value_counts(dropna=False).to_dict()
   113      metadata: MutableMapping[str, Any] = {
   114          "method": method,
   115          "edges": cat.cat.categories.left.tolist() + [cat.cat.categories.right[-1]],
   116          "labels": labels,
   117          "counts": counts,
   118      }
   119      if method != "edges":
   120          metadata["labels"] = _format_interval(np.asarray(metadata["edges"]), precision)
   121          cat = cat.cat.rename_categories(metadata["labels"])
   122      return cat, dict(metadata)

   125  def cross_agg(
   126      df: pd.DataFrame,
   127      x_spec: BinSpec,
   128      y_spec: BinSpec,
   129      value: str,
   130      agg: Literal["count", "sum", "mean", "rate"] = "rate",
   131  ) -> Tuple[pd.DataFrame, Dict[str, Any]]:
   132      """Aggregate ``value`` over the 2D grid defined by ``x_spec`` and ``y_spec``.
   133
   134      The returned DataFrame contains columns ``x_bin``, ``y_bin``, ``count`` and
   135      ``metric`` (either rate/mean/sum as requested). Metadata includes the bin
   136      definitions for downstream visualization labels.
   137      """
   138
   139      x_binned, x_info = bin_series(
   140          df[x_spec.column], x_spec.method, x_spec.bins, precision=x_spec.precision
   141      )
   142      y_binned, y_info = bin_series(
   143          df[y_spec.column], y_spec.method, y_spec.bins, precision=y_spec.precision
   144      )
   145      tmp = df.copy()
   146      tmp["x_bin"] = x_binned
   147      tmp["y_bin"] = y_binned
   148
   149      grouped = tmp.dropna(subset=["x_bin", "y_bin"])
   150      if agg == "count":
   151          agg_series = grouped.groupby(["x_bin", "y_bin"])[value].count()
   152      elif agg == "sum":
   153          agg_series = grouped.groupby(["x_bin", "y_bin"])[value].sum()
   154      else:
   155          agg_series = grouped.groupby(["x_bin", "y_bin"])[value].mean()
   156
   157      counts = grouped.groupby(["x_bin", "y_bin"])[value].count()
   158      result = (
   159          agg_series.rename("metric")
   160          .to_frame()
   161          .join(counts.rename("count"))
   162          .reset_index()
   163      )
   164
   165      if agg == "rate":
   166          result.rename(columns={"metric": "rate"}, inplace=True)
   167      elif agg == "mean":
   168          result.rename(columns={"metric": "mean"}, inplace=True)
   169      else:
   170          result.rename(columns={"metric": agg}, inplace=True)
   171
   172      metadata = {"x": x_info, "y": y_info, "agg": agg, "value": value}
   173      return result, metadata

    20  def _cliffs_delta(x: np.ndarray, y: np.ndarray) -> float:
    21      """Compute Cliff's delta effect size."""
    22
    23      x = np.asarray(x)
    24      y = np.asarray(y)
    25      n_x = x.size
    26      n_y = y.size
    27      if n_x == 0 or n_y == 0:
    28          return np.nan
    29      pairwise = np.subtract.outer(x, y)
    30      greater = np.sum(pairwise > 0)
    31      less = np.sum(pairwise < 0)
    32      return (greater - less) / (n_x * n_y)

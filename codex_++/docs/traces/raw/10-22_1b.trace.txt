## 2 Run/Drift Cadence

> You must detect and adapt to cognitive cadence according to user requests and prompts, as well as pattern-recognition anticipa
tion.

* **Run Mode:**
* Prioritize execution, implementation, and velocity.
* Minimize unsolicited design analysis.
* Leave `.intent.md` trails for future Drift review.
* **Drift Mode:**
* Reflect, consolidate `.intent.md` into `.purpose.md`.
* Validate against system assumptions, purpose alignment, and past commits.
* Suggest design refactor, dependency graph updates, or tension annotation.

If uncertain, default to **Drift**.

Cadence tags may be used for clarity:

* @ai-cadence: run
* @ai-cadence: drift

@dataclass
class FunctionSignature:
    """Simplified function signature extracted from the AST."""

    name: str
    parameters: List[str]
    return_annotation: Optional[str]

    @classmethod
    def from_function_def(cls, node: ast.AST) -> "FunctionSignature":
        if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            raise TypeError("Expected FunctionDef/AsyncFunctionDef node")

        args = node.args
        params: List[str] = []
        for arg in list(args.posonlyargs) + list(args.args):
            params.append(arg.arg)
        if args.vararg:
            params.append(f"*{args.vararg.arg}")
        for kw in args.kwonlyargs:
            params.append(f"{kw.arg}")
        if args.kwarg:
            params.append(f"**{args.kwarg.arg}")

        annotation = _annotation_to_str(node.returns)
        return cls(name=node.name, parameters=params, return_annotation=annotation)


@dataclass
class MethodSignature(FunctionSignature):
    """Marker class for readability."""


@dataclass
class ClassSignature:
    """Simplified representation of a class definition."""

    name: str
    methods: Dict[str, MethodSignature]

    @classmethod
    def from_class_def(cls, node: ast.ClassDef) -> "ClassSignature":
        methods: Dict[str, MethodSignature] = {}
        for item in node.body:
            if isinstance(item, (ast.FunctionDef, ast.AsyncFunctionDef)):
                signature = FunctionSignature.from_function_def(item)
                # Ignore implicit ``self`` when comparing method parameters.
                if signature.parameters and signature.parameters[0] in {"self", "cls"}:
                    parameters = signature.parameters[1:]
                else:
                    parameters = signature.parameters
                methods[item.name] = MethodSignature(
                    name=signature.name,
                    parameters=parameters,
                    return_annotation=signature.return_annotation,
                )
        return cls(name=node.name, methods=methods)


def _annotation_to_str(annotation: Optional[ast.AST]) -> Optional[str]:
    if annotation is None:
        return None
    return ast.unparse(annotation) if hasattr(ast, "unparse") else None

def inspect_module(path: Path) -> Tuple[Dict[str, FunctionSignature], Dict[str, ClassSignature]]:
    module_ast = ast.parse(Path(path).read_text())
    functions: Dict[str, FunctionSignature] = {}
    classes: Dict[str, ClassSignature] = {}
    for node in module_ast.body:
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            signature = FunctionSignature.from_function_def(node)
            functions[signature.name] = signature
        elif isinstance(node, ast.ClassDef):
            classes[node.name] = ClassSignature.from_class_def(node)
    return functions, classes

> def _cohen_d(x: Sequence[float], y: Sequence[float]) -> float:
>     x = np.asarray(x, dtype=float)
>     y = np.asarray(y, dtype=float)
>     nx, ny = len(x), len(y)
>     if nx < 2 or ny < 2:
>         return float("nan")
>     vx = x.var(ddof=1)
>     vy = y.var(ddof=1)
>     pooled = ((nx - 1) * vx + (ny - 1) * vy) / (nx + ny - 2)
>     if pooled <= 0:
>         return 0.0
>     return (x.mean() - y.mean()) / np.sqrt(pooled)

def _ensure_dir(path: Path) -> Path:
    path.mkdir(parents=True, exist_ok=True)
    return path


def _write_table(df: pd.DataFrame, path: Path) -> str:
    path.parent.mkdir(parents=True, exist_ok=True)
    df.to_csv(path, index=False)
    return str(path)


def build_report(df: pd.DataFrame, config: Mapping[str, Any]) -> Dict[str, Any]:
    output_dir = Path(config.get("output_dir", DEFAULT_OUTPUT_DIR))
    image_formats = tuple(config.get("image_formats", ("png", "svg")))
    tables_dir = _ensure_dir(output_dir / "tables")

    manifest: Dict[str, Any] = {}
    modules_cfg = config.get("modules", {})
    readmit_col = config.get("readmit_col", READMIT_COL)

    if modules_cfg.get("bars_hists", {}).get("enabled", True):
        cfg = modules_cfg.get("bars_hists", {})
        result = build_bars_hists(
            df,
            categoricals=cfg.get("categoricals", []),
            numerics=cfg.get("numerics", []),
            readmit_col=cfg.get("readmit_col", readmit_col),
        )
        figure_paths = save_figure(result.figure, "bars_hists", _ensure_dir(output_dir), formats=image_formats)
        table_path = _write_table(result.summary, tables_dir / "bars_hists_summary.csv")
        plt.close(result.figure)
        manifest["bars_hists"] = {
            "figure": figure_paths,
            "table": table_path,
            "notes": "Bar charts for categoricals and histograms for numerics.",
        }

    if modules_cfg.get("boxplots", {}).get("enabled", True):
        cfg = modules_cfg.get("boxplots", {})
        result = build_boxplots(
            df,
            numerics=cfg.get("numerics", []),
            readmit_col=cfg.get("readmit_col", readmit_col),
        )
        figure_paths = save_figure(result.figure, "boxplots", _ensure_dir(output_dir), formats=image_formats)
        table_path = _write_table(result.stats, tables_dir / "boxplots_stats.csv")
        plt.close(result.figure)
        manifest["boxplots"] = {
            "figure": figure_paths,
            "table": table_path,
            "notes": "Box/violin plots with effect sizes.",
        }

    if modules_cfg.get("corrgram", {}).get("enabled", True):
        cfg = modules_cfg.get("corrgram", {})
        result = build_corrgram(
            df,
            numerics=cfg.get("numerics", []),
            readmit_col=cfg.get("readmit_col", readmit_col),
            sample=cfg.get("sample"),
        )
        figure_paths = save_figure(result.figure, "corrgram", _ensure_dir(output_dir), formats=image_formats)
        table_path = _write_table(result.metrics, tables_dir / "corrgram_metrics.csv")
        plt.close(result.figure)
        manifest["corrgram"] = {
            "figure": figure_paths,
            "table": table_path,
            "notes": "3x3 corrgram metrics and scatter panels.",
        }

    if modules_cfg.get("parallel_coords", {}).get("enabled", True):
        cfg = modules_cfg.get("parallel_coords", {})
        result = build_parallel_coords(
            df,
            cols=cfg.get("cols", []),
            readmit_col=cfg.get("readmit_col", readmit_col),
            diagnosis_col=cfg.get("diagnosis_col", DEFAULT_DIAG_COL),
            top_k_diag=cfg.get("top_k_diag", 6),
            scaling=cfg.get("scaling", "raw"),
        )
        plot_dir = _ensure_dir(output_dir)
        html_path = plot_dir / "parallel_coords.html"
        result.figure.write_html(html_path)
        manifest["parallel_coords"] = {
            "figure": {"html": str(html_path)},
            "notes": "Interactive parallel coordinates with diagnosis selector.",
        }

    if modules_cfg.get("sankey", {}).get("enabled", True):
        cfg = modules_cfg.get("sankey", {})
        result = build_sankey(
            df,
            top_k=cfg.get("top_k", 10),
            min_count=cfg.get("min_count", 25),
            palette=cfg.get("palette"),
        )
        plot_dir = _ensure_dir(output_dir)
        html_path = plot_dir / "sankey.html"
        result.figure.write_html(html_path)
        manifest["sankey"] = {

    return PlotOutput(figure=fig, summary=summary)

def _heatmap_panels(
    df: pd.DataFrame,
    diagnoses: Sequence[str],
    bin_specs: Mapping[str, Mapping[str, object]],
    readmit_col: str,
) -> Tuple[plt.Figure, pd.DataFrame]:
    apply_base_theme()
    prior_bins = _bin_column(df, PRIOR_ADM_COL, bin_specs[PRIOR_ADM_COL])
    df = df.assign(prior_bin=prior_bins)

    ncols = 3
    nrows = max(1, math.ceil(len(diagnoses) / ncols))
    fig, axes = plt.subplots(nrows, ncols, figsize=(5 * ncols, 4 * nrows))
    axes_list = _flatten_axes(axes)
    tables: List[pd.DataFrame] = []

    for idx, diag in enumerate(diagnoses):
        ax = axes_list[idx]
        subset = df[df[DIAG_COL] == diag]
        pivot = (
            subset.pivot_table(
                values=readmit_col,
                index="prior_bin",
                columns=DISCHARGE_COL,
                aggfunc="mean",
            )
            .fillna(0)
        )
        tidy = (
            pivot.reset_index()
            .melt(id_vars="prior_bin", var_name=DISCHARGE_COL, value_name="rate")
            .assign(diagnosis=diag)
        )
        tables.append(tidy)
        sns.heatmap(pivot, annot=True, fmt=".2f", cmap="BuGn", vmin=0, vmax=1, ax=ax)
        title_with_subtitle(ax, diag, subtitle="Prior Admissions × Discharge", base_rate=float(df[readmit_col].mean()))
        ax.set_xlabel(DISCHARGE_COL)
        ax.set_ylabel("Prior Admissions bin")

    for ax in axes_list[len(diagnoses) :]:
        ax.remove()

    table = pd.concat(tables, ignore_index=True) if tables else pd.DataFrame()
    fig.tight_layout()
    return fig, table

def _cohen_d(x: Sequence[float], y: Sequence[float]) -> float:
    x = np.asarray(x, dtype=float)
    y = np.asarray(y, dtype=float)
    nx, ny = len(x), len(y)
    if nx < 2 or ny < 2:
        return float("nan")
    vx = x.var(ddof=1)
    vy = y.var(ddof=1)
    pooled = ((nx - 1) * vx + (ny - 1) * vy) / (nx + ny - 2)
    if pooled <= 0:
        return 0.0
    return (x.mean() - y.mean()) / np.sqrt(pooled)


def _mannwhitney(x: Sequence[float], y: Sequence[float]) -> float:
    try:
        stat, pvalue = stats.mannwhitneyu(x, y, alternative="two-sided")
    except ValueError:
        return float("nan")
    return float(pvalue)

    28  def bin_series(
    29      series: pd.Series,
    30      method: BinMethod,
    31      bins: int | Sequence[float],
    32      include_lowest: bool = True,
    33      precision: int = 2,
    34  ) -> Tuple[pd.Categorical, Dict[str, Any]]:
    35      """Bin ``series`` using the provided strategy."""
    36
    37      clean = series.dropna()
    38      if clean.empty:
    39          categories = pd.Categorical([pd.NA] * len(series))
    40          return categories, {"edges": [], "counts": pd.Series(dtype=int)}
    41
    42      if method == "equal":
    43          cat = pd.cut(series, bins=int(bins), include_lowest=include_lowest, precision=precision)
    44          edges = cat.cat.categories.left.tolist() + [cat.cat.categories.right[-1]]
    45      elif method == "quantile":
    46          cat = pd.qcut(
    47              series,
    48              q=int(bins),
    49              duplicates="drop",
    50              precision=precision,
    51          )
    52          edges = cat.cat.categories.left.tolist() + [cat.cat.categories.right[-1]]
    53      elif method == "edges":
    54          edges = list(bins)
    55          cat = pd.cut(series, bins=edges, include_lowest=include_lowest, precision=precision)
    56      else:
    57          raise ValueError(f"Unsupported binning method: {method}")
    58
    59      info = {
    60          "edges": edges,
    61          "counts": cat.value_counts().sort_index(),
    62      }
    63      return cat, info

    66  def cross_agg(
    67      df: pd.DataFrame,
    68      xbin: str,
    69      ybin: str,
    70      value: str,
    71      agg: Literal["count", "mean"],
    72  ) -> pd.DataFrame:
    73      """Aggregate ``df`` into a 2D table based on pre-computed bins."""
    74
    75      grouped = df.groupby([xbin, ybin])[value]
